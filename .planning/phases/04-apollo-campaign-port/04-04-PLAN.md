Apolllo `/import` writes to the `companies` table per imported person. If two parallel imports hit the same `apolloOrgId` simultaneously, Prisma throws `P2002` on the unique constraint. The fix: 3-case ladder.

If 04-01 Task 4 already included this from `861ed00`, verify and skip. Else:

In `routes/apollo.ts`, find the `prisma.company.create({ ... })` call in the import loop. Wrap in try/catch:

```typescript
async function upsertCompany(prisma: PrismaClient, org: ApolloOrg) {
  // Case 1: find by apolloOrgId
  const existingByApollo = await prisma.company.findUnique({ where: { apolloOrgId: org.id } });
  if (existingByApollo) {
    return prisma.company.update({
      where: { id: existingByApollo.id },
      data: { domain: org.domain ?? existingByApollo.domain, name: org.name, apolloRawData: org as any },
    });
  }
  // Case 2: find by domain (if domain present)
  if (org.domain) {
    const existingByDomain = await prisma.company.findUnique({ where: { domain: org.domain } });
    if (existingByDomain) {
      return prisma.company.update({
        where: { id: existingByDomain.id },
        data: { apolloOrgId: org.id, apolloRawData: org as any },
      });
    }
  }
  // Case 3: create
  try {
    return await prisma.company.create({
      data: { name: org.name, apolloOrgId: org.id, domain: org.domain, apolloRawData: org as any },
    });
  } catch (e: any) {
    if (e.code === 'P2002') {
      // Race: someone else just created it. Re-fetch.
      const racedByApollo = await prisma.company.findUnique({ where: { apolloOrgId: org.id } });
      if (racedByApollo) return racedByApollo;
      if (org.domain) {
        const racedByDomain = await prisma.company.findUnique({ where: { domain: org.domain } });
        if (racedByDomain) return racedByDomain;
      }
    }
    throw e;
  }
}
```

```typescript
import Anthropic from '@anthropic-ai/sdk';

const anthropic = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });

export async function normalizeFiltersWithClaude(rawInput: {
  titles?: string;
  industries?: string;
  keywords?: string;
}): Promise<{
  job_titles: string[];
  industries: string[];
  technologies: string[];
  warnings: string[];
}> {
  const prompt = `You are an Apollo.io ICP refiner. The user typed these search terms:
Titles: ${rawInput.titles || '(none)'}
Industries: ${rawInput.industries || '(none)'}
Keywords: ${rawInput.keywords || '(none)'}

Normalize into Apollo.io's filter taxonomy. Common pitfalls:
- "NetSuite" as a keyword returns consultancies, not customers. Suggest tech filter "uses_netsuite" instead.
- "AI" returns startups. Use specific titles like "ML Engineer", "Data Scientist".

Return JSON: { job_titles, industries, technologies, warnings }.`;

  const response = await anthropic.messages.create({
    model: 'claude-haiku-4-5-20251001',
    max_tokens: 1024,
    messages: [{ role: 'user', content: prompt }],
  });

  const text = (response.content[0] as any).text;
  const jsonMatch = text.match(/\{[\s\S]*\}/);
  if (!jsonMatch) throw new Error('Claude returned non-JSON');
  return JSON.parse(jsonMatch[0]);
}
```
