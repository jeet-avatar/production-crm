```typescript
// backend/src/services/scheduledDispatcher.ts
import { PrismaClient } from '@prisma/client';
import { Resend } from 'resend';

const POLL_INTERVAL_MS = 30_000; // 30s
let pollerHandle: NodeJS.Timeout | null = null;

// Re-uses the same Sara constant from apollo.ts — DO NOT introduce a different sender.
const APOLLO_FROM_EMAIL = 'Sara <sara@techcloudpro.com>';
const RESEND_API_KEY = process.env.RESEND_API_KEY!;
if (!RESEND_API_KEY) {
  console.error('[scheduledDispatcher] FATAL: RESEND_API_KEY missing.');
  process.exit(1);
}
const resend = new Resend(RESEND_API_KEY);
const prisma = new PrismaClient();

export async function dispatchDueScheduled() {
  const due = await prisma.emailLog.findMany({
    where: {
      status: 'SCHEDULED',
      // scheduledAt <= now() — need to add scheduledAt column to EmailLog if not present
      scheduledAt: { lte: new Date() },
    },
    take: 50, // safety cap per cycle
    orderBy: { scheduledAt: 'asc' },
  });

  for (const row of due) {
    try {
      // Send via Resend with the SAME hardcoded Sara sender + reply-to
      const result = await resend.emails.send({
        from: APOLLO_FROM_EMAIL,
        to: row.toEmail!,
        subject: (row.metadata as any)?.subject ?? '(scheduled send)',
        html: (row.metadata as any)?.html ?? '<p>(missing body)</p>',
        reply_to: 'sara@techcloudpro.com',
      });

      await prisma.emailLog.update({
        where: { id: row.id },
        data: {
          status: 'SENT',
          sentAt: new Date(),
          messageId: result.data?.id ?? null,
          fromEmail: 'sara@techcloudpro.com',
        },
      });

      console.log(`[scheduledDispatcher] dispatched ${row.id} → ${row.toEmail}`);
    } catch (e: any) {
      await prisma.emailLog.update({
        where: { id: row.id },
        data: { status: 'FAILED', errorMessage: e.message ?? 'unknown' },
      });
      console.error(`[scheduledDispatcher] FAILED ${row.id}: ${e.message}`);
    }
  }

  return due.length;
}

export function startScheduledDispatcher() {
  if (pollerHandle) {
    console.warn('[scheduledDispatcher] already running, ignoring duplicate start');
    return;
  }

  // Boot catch-up: handle anything that became due during downtime
  dispatchDueScheduled()
    .then(n => console.log(`[scheduledDispatcher] boot catch-up dispatched ${n} due rows`))
    .catch(e => console.error('[scheduledDispatcher] boot catch-up error:', e));

  // Steady-state poller
  pollerHandle = setInterval(() => {
    dispatchDueScheduled().catch(e =>
      console.error('[scheduledDispatcher] poll error:', e)
    );
  }, POLL_INTERVAL_MS);

  console.log(`[scheduledDispatcher] started; polling every ${POLL_INTERVAL_MS / 1000}s`);
}

export function stopScheduledDispatcher() {
  if (pollerHandle) {
    clearInterval(pollerHandle);
    pollerHandle = null;
    console.log('[scheduledDispatcher] stopped');
  }
}
```

          },
        });
      }));
      return res.json({ scheduled: true, scheduledAt: sendTime, count: rows.length });
