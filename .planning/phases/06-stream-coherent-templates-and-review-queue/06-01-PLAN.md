    STEP E — Rewrite `upgradeStreamTemplatesToV3` to be stream-aware. Replace the existing function body (around lines 577-602) with:

    ```ts
    /**
     * Idempotent in-place upgrade of the 9 Stream:* email_templates rows from v2 → v3.
     * Stream-aware: looks up STREAM_TEMPLATE_V3_BODIES[row.category] per row, falls back
     * to STREAM_TEMPLATE_V3_BODIES['Stream: Other'] if the category is not in the dict.
     *
     * Idempotency sentinel is also stream-aware: each per-stream body contains a UNIQUE
     * `metricLabel` literal (e.g., "Cybersecurity Practice", "Automation Practice", etc.).
     * A row whose htmlContent already contains the metricLabel for its category short-circuits.
     */
    export async function upgradeStreamTemplatesToV3(
      prisma: PrismaClient,
      userId: string,
    ): Promise<{ upgraded: string[]; alreadyV3: string[]; total: number }> {
      // Per-stream metricLabel sentinels — must match the buildStreamV3Body opts above.
      const STREAM_V3_SENTINELS: Record<string, string> = {
        'Stream: Cybersecurity': 'Cybersecurity Practice',
        'Stream: NetSuite': 'NetSuite Practice',
        'Stream: RPA': 'Automation Practice',
        'Stream: Web3': 'Web3 Practice',
        'Stream: Data': 'Data Practice',
        'Stream: Cloud': 'Cloud Practice',
        'Stream: Mobile': 'Mobile Practice',
        'Stream: AI-ML': 'AI/ML Practice',
        'Stream: Other': 'TCP Practice',
      };

      const rows = await prisma.emailTemplate.findMany({
        where: { userId, category: { startsWith: 'Stream:' } },
        select: { id: true, name: true, category: true, htmlContent: true },
      });

      const upgraded: string[] = [];
      const alreadyV3: string[] = [];

      for (const row of rows) {
        const sentinel = STREAM_V3_SENTINELS[row.category] ?? STREAM_V3_SENTINELS['Stream: Other'];
        if ((row.htmlContent || '').includes(sentinel)) {
          alreadyV3.push(row.name);
          continue;
        }
        const body = STREAM_TEMPLATE_V3_BODIES[row.category] ?? STREAM_TEMPLATE_V3_BODIES['Stream: Other'];
        await prisma.emailTemplate.update({
          where: { id: row.id },
          data: { htmlContent: body },
        });
        upgraded.push(row.name);
      }

      return { upgraded, alreadyV3, total: rows.length };
    }
    ```
