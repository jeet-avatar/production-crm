```tsx
// frontend/src/components/NetSuiteCampaignWizard.tsx
import { useEffect, useState } from 'react';
import { useNavigate } from 'react-router-dom';
import {
  CheckCircleIcon,
  SparklesIcon,
  XMarkIcon,
} from '@heroicons/react/24/outline';
import {
  contactsApi,
  emailTemplatesApi,
  apolloApi,
  campaignsApi,    // ONLY for aiGenerateContent — NOT for sending
} from '../services/api';

interface Contact {
  id: string;
  firstName?: string | null;
  lastName?: string | null;
  email: string;
  stream?: string | null;
  companyId?: string | null;
  company?: { id: string; name: string } | null;
}

interface NetSuiteCampaignWizardProps {
  isOpen: boolean;
  onClose: () => void;
  importedContactIds: string[];
  suggestedStream: string;
  onSuccess?: () => void;
}

const HARDCODED_FALLBACK = {
  subject: 'Tech engineers available for {{companyName}}',
  htmlBody: '<p>Hi {{firstName}},</p><p>We place senior engineers across the stack. Quick chat about {{companyName}}\'s near-term needs?</p><p>— Sara, TechCloudPro</p>',
};

export function NetSuiteCampaignWizard({
  isOpen, onClose, importedContactIds, suggestedStream, onSuccess,
}: NetSuiteCampaignWizardProps) {
  const navigate = useNavigate();
  const [step, setStep] = useState<1 | 2 | 3 | 4>(1);
  const [contacts, setContacts] = useState<Contact[]>([]);
  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
  const [subject, setSubject] = useState('');
  const [body, setBody] = useState('');
  const [templateId, setTemplateId] = useState<string | null>(null);
  const [usingFallback, setUsingFallback] = useState(false);
  const [aiLoading, setAiLoading] = useState(false);
  const [aiError, setAiError] = useState<string | null>(null);
  const [sentCount, setSentCount] = useState(0);
  const [failedCount, setFailedCount] = useState(0);
  const [sending, setSending] = useState(false);
  const [sendError, setSendError] = useState<string | null>(null);

  // 1. On mount: fetch contacts + lookup template
  useEffect(() => {
    if (!isOpen) return;
    let cancelled = false;
    (async () => {
      // Fetch contacts
      try {
        const { contacts } = await contactsApi.getByIds(importedContactIds);
        if (cancelled) return;
        setContacts(contacts);
        setSelectedIds(new Set(contacts.map((c: Contact) => c.id)));  // pre-tick all
      } catch (e) {
        console.error('[wizard] fetch contacts failed', e);
      }
      // 3-layer template fallback
      const fetchTemplate = async (category: string) => {
        try { return await emailTemplatesApi.findByCategory(category); } catch { return null; }
      };
      let tpl = await fetchTemplate(`Stream:${suggestedStream}`);
      if (!tpl) tpl = await fetchTemplate(`Stream:Other`);
      if (cancelled) return;
      if (tpl) {
        setSubject(tpl.subject || '');
        setBody(tpl.htmlBody || tpl.body || '');
        setTemplateId(tpl.id);
        setUsingFallback(false);
      } else {
        setSubject(HARDCODED_FALLBACK.subject);
        setBody(HARDCODED_FALLBACK.htmlBody);
        setTemplateId(null);
        setUsingFallback(true);
      }
    })();
    return () => { cancelled = true; };
  }, [isOpen, importedContactIds, suggestedStream]);

  if (!isOpen) return null;

  // STEP 1: contact picklist
  // STEP 2: subject + body editor + "Let AI write it ✨" button
  // STEP 3: preview + live validation (subject empty / body <50 / 0 selected)
  // STEP 4: green success ("Sent to N companies via Resend")

  // ===== Step 2 AI generation =====
  async function handleAiGenerate() {
    setAiLoading(true);
    setAiError(null);
    try {
      const data = await campaignsApi.aiGenerateContent(
        `${suggestedStream} outreach`,
        'professional',
        contacts[0]?.company?.name || 'your team',
      );
      setBody(data.content || body);
    } catch (e: any) {
      setAiError('AI write failed — try again or write it yourself.');
    } finally {
      setAiLoading(false);
    }
  }

  // ===== Step 3 send (USER-LOCKED 2026-05-30: Resend via /api/apollo/send-campaign) =====
  // We do NOT call campaignsApi.create/addCompany/send (that hits campaigns.ts SES — Rajesh's path).
  // We do NOT create a Campaign DB row in Phase 4 MVP.
  // We POST contactIds + templateId + suggestedStream to the new endpoint and let backend Resend-dispatch.
  async function handleSend() {
    if (!templateId) {
      // Edge case: user is on hardcoded fallback (no DB-backed template).
      // Phase 4 MVP requirement: templateId is required by /api/apollo/send-campaign — surface a clear error.
      setSendError(
        'No saved template found for this stream. Go to Email Templates and seed templates, ' +
        'or edit/save this draft as a template first.',
      );
      return;
    }
    setSending(true);
    setSendError(null);
    try {
      const selectedContactIds = contacts.filter((c) => selectedIds.has(c.id)).map((c) => c.id);

      // POST to the NEW Phase 4 Resend backend (plan 04-03 Task 3).
      const result = await apolloApi.sendCampaign(
        selectedContactIds,
        templateId,
        suggestedStream,
      );

      setSentCount(result.sent);
      setFailedCount(result.failed);
      setStep(4);
      onSuccess?.();
    } catch (e: any) {
      const msg = e?.response?.data?.error || e?.message || 'send failed';
      console.error('[wizard] send failed', e);
      setSendError(msg);
    } finally {
      setSending(false);
    }
  }

  // ===== Step 3 validation =====
  const validationWarnings: string[] = [];
  if (!subject.trim()) validationWarnings.push('Your email has no subject — add one in Step 2');
  if (body.replace(/<[^>]+>/g, '').trim().length < 50) validationWarnings.push('Your message is very short — consider adding more detail');
  if (selectedIds.size === 0) validationWarnings.push('You haven\'t picked any companies — go back to Step 1');
  const sendBlocked = !subject.trim() || body.replace(/<[^>]+>/g, '').trim().length < 10 || selectedIds.size === 0;

  return (
    <div /* modal overlay */>
      <div /* modal box */>
        {/* close button */}
        {/* 4-dot step indicator: 1.Audience 2.Email 3.Review 4.Done */}

        {step === 1 && (
          <section>
            <h2>Pick the contacts you want to email</h2>
            <p>These are the {contacts.length} contact(s) you just imported from Apollo. Suggested stream: <strong>{suggestedStream}</strong></p>
            {/* Select All checkbox + list of contacts (checkbox + firstName lastName + email + company name) */}
            <p>{selectedIds.size} of {contacts.length} selected</p>
            <button disabled={selectedIds.size === 0} onClick={() => setStep(2)}>Next: Write your email →</button>
          </section>
        )}

        {step === 2 && (
          <section>
            <h2>Write your email</h2>
            {usingFallback && (
              <div style={{ background: '#fff3cd', padding: '12px', borderRadius: '8px' }}>
                Using built-in template — go to Email Templates to customize per-stream copy.
                (Sending requires a saved template — seed them first via the Email Templates page.)
              </div>
            )}
            <label>📝 The title of your email</label>
            <input value={subject} onChange={(e) => setSubject(e.target.value)} />
            <label>✍️ What you want to say</label>
            <textarea value={body} onChange={(e) => setBody(e.target.value)} rows={12} />
            <p>{body.length} characters. Variables {`{{firstName}}`} + {`{{companyName}}`} substitute per-recipient on the server.</p>
            <button onClick={handleAiGenerate} disabled={aiLoading}>
              <SparklesIcon /> {aiLoading ? 'Generating…' : 'Let AI write it for me ✨'}
            </button>
            {aiError && <p style={{ color: 'red' }}>{aiError}</p>}
            <button onClick={() => setStep(1)}>← Back</button>
            <button onClick={() => setStep(3)}>Next: Review →</button>
          </section>
        )}

        {step === 3 && (
          <section>
            <h2>Review before sending</h2>
            {/* email preview card */}
            <div>
              <div><strong>From:</strong> Sara &lt;sara@techcloudpro.com&gt;</div>
              <div><strong>Subject:</strong> {subject || '(none)'}</div>
              <div dangerouslySetInnerHTML={{ __html: body }} />
            </div>
            {validationWarnings.map((w, i) => (
              <div key={i} style={{ background: '#fff3cd', padding: '8px' }}>⚠ {w}</div>
            ))}
            {sendError && (
              <div style={{ background: '#fee2e2', color: '#991b1b', padding: '12px', borderRadius: '8px' }}>
                ❌ {sendError}
              </div>
            )}
            <p>Ready to send to {selectedIds.size} contact(s) via Resend</p>
            <button onClick={() => setStep(2)}>← Back</button>
            <button disabled={sendBlocked || sending} onClick={handleSend}>
              {sending ? 'Sending…' : `Send ${selectedIds.size} email(s) now →`}
            </button>
          </section>
        )}

        {step === 4 && (
          <section style={{ textAlign: 'center' }}>
            <CheckCircleIcon style={{ width: 64, height: 64, color: '#10b981' }} />
            <h2>Your emails are on their way! ✅</h2>
            <p>Sent {sentCount} of {sentCount + failedCount} via Resend.</p>
            {failedCount > 0 && (
              <p style={{ color: '#b45309' }}>{failedCount} failed — check console for details</p>
            )}
            <button onClick={() => { onClose(); navigate('/contacts'); }}>Go see your contacts →</button>
            <button onClick={() => { setStep(1); setSentCount(0); setFailedCount(0); setSendError(null); }}>Send another campaign</button>
          </section>
        )}
      </div>
    </div>
  );
}
```
